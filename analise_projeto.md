import os
import xml.etree.ElementTree as ET

PASTA_XML = r"C:\xml_esocial"

def ler_xml(caminho):
    try:
        tree = ET.parse(caminho)
        return tree.getroot()
    except Exception as e:
        print(f"Erro ao ler {caminho}: {e}")
        return None

def extrair_dados(root):
    ns = {
        'ns': 'http://www.esocial.gov.br/schema/evt/evtIrrfBenef/v_S_01_03_00'
    }

    dados = {}

    try:
        dados['cpf'] = root.find('.//ns:cpfBenef', ns).text
        dados['perApur'] = root.find('.//ns:perApur', ns).text
        dados['vlrRendTrib'] = float(root.find('.//ns:vlrRendTrib', ns).text)
        dados['vlrPrev'] = float(root.find('.//ns:vlrPrevOficial', ns).text)
        dados['vlrIR'] = float(root.find('.//ns:vlrCRMen', ns).text)
    except:
        return None

    return dados

def validar(dados):
    erros = []

    base = dados['vlrRendTrib'] - dados['vlrPrev']

    if base <= 0:
        erros.append("Base IR inválida")

    if dados['vlrIR'] < 0:
        erros.append("IR negativo")

    return erros

def processar():
    for raiz, _, arquivos in os.walk(PASTA_XML):
        for arq in arquivos:
            if arq.endswith(".xml"):
                caminho = os.path.join(raiz, arq)

                root = ler_xml(caminho)
                if not root:
                    continue

                dados = extrair_dados(root)
                if not dados:
                    continue

                erros = validar(dados)

                print("\nArquivo:", arq)
                print(dados)

                if erros:
                    print("❌ Erros:", erros)
                else:
                    print("✅ OK")

if __name__ == "__main__":
    processar()

def parse_s5000(root, arquivo, empresa):
    dados = []

    try:
        # pega namespace dinamicamente
        ns = {'ns': root.tag.split('}')[0].strip('{')}

        cpf = root.find('.//ns:cpfTrab', ns)
        per = root.find('.//ns:perApur', ns)

        if cpf is None or per is None:
            return []

        cpf = cpf.text
        comp = per.text

        # percorre bases
        for ide in root.findall('.//ns:ideEstabLot', ns):
            for bases in ide.findall('.//ns:basesRemun', ns):

                cod = bases.find('ns:codRubr', ns)
                vr_base = bases.find('ns:vrBcCp00', ns)
                vr_inss = bases.find('ns:vrCpSeg', ns)
                vr_ir = bases.find('ns:vrBcIrrf', ns)

                dados.append((
                    cpf,
                    empresa,
                    comp,
                    cod.text if cod is not None else None,
                    float(vr_base.text) if vr_base is not None else 0,
                    float(vr_inss.text) if vr_inss is not None else 0,
                    float(vr_ir.text) if vr_ir is not None else 0,
                    arquivo
                ))

    except Exception as e:
        print("Erro S5000:", arquivo, e)

    return dados

def extrair_evento(root):
    eventos = root.findall('.//{*}retornoProcessamentoDownload/{*}evento')

    for ev in eventos:
        esocial = ev.find('.//{*}eSocial')
        if esocial is None:
            continue

        for child in list(esocial):
            tag = child.tag.lower()

            if 'evtirrfbenef' in tag:
                return 'S1210', child

            if 'evtbasestrab' in tag:
                return 'S5001', child

            if 'evtirrf' in tag and 'benef' not in tag:
                return 'S5002', child

            if 'evtbasesfgts' in tag:
                return 'S5003', child

            if 'evtbases' in tag and 'trab' not in tag:
                return 'S5011', child

    return None, None

def parse_s1210(node, empresa, arq):

    except:
        return None


def parse_s5001(node, empresa, arq):
    ns = {'ns': node.tag.split('}')[0].strip('{')}
    dados = []

    cpf = node.find('.//ns:cpfTrab', ns)
    comp = node.find('.//ns:perApur', ns)

    if cpf is None:
        return []

    for info in node.findall('.//ns:infoCpCalc', ns):
        inss = info.find('ns:vrCpSeg', ns)
        desc = info.find('ns:vrDescSeg', ns)

        dados.append((
            cpf.text,
            empresa,
            comp.text,
            float(inss.text) if inss is not None else 0,
            float(desc.text) if desc is not None else 0,
            arq
        ))

    return dados
